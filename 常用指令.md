docker compose exec -T backend bash -lc '
cd /app/backend && /app/.venv/bin/python  - << "PY"
from sqlmodel import Session
from app.core.db import engine
from app import crud
from app.models import UserCreate

users = [
    {"email": "admin_demo@example.com", "password": "Admin123456", "is_superuser": True, "is_active": True, "full_name": "Demo Admin"},
    {"email": "user_demo1@example.com", "password": "User123456", "is_superuser": False, "is_active": True, "full_name": "Demo User 1"},
    {"email": "user_demo2@example.com", "password": "User123456", "is_superuser": False, "is_active": True, "full_name": "Demo User 2"},
    {"email": "disabled_demo@example.com", "password": "User123456", "is_superuser": False, "is_active": False, "full_name": "Disabled Demo"},
]

with Session(engine) as session:
    for u in users:
        existing = crud.get_user_by_email(session=session, email=u["email"])
        if existing:
            print(f"SKIP  {u['email']} (already exists)")
            continue

        user_in = UserCreate(
            email=u["email"],
            password=u["password"],
            is_superuser=u.get("is_superuser", False),
            is_active=u.get("is_active", True),
            full_name=u.get("full_name"),
        )
        created = crud.create_user(session=session, user_create=user_in)
        print(f"CREATED {created.email} id={created.id}")
PY'
