### 5.2 示例 API 封装

// api/flows.ts
import request from './index'

export const flowApi = {
  // 获取流量列表
  getList(interfaceId: string, params: FlowListParams) {
    return request.get(`/interfaces/${interfaceId}/flows`, { params })
  },

  // 获取流量详情
  getDetail(flowId: string) {
    return request.get(`/flows/${flowId}`)
  },

  // 批量打标签
  batchTag(data: { flow_ids: string[], tags: string[] }) {
    return request.post('/flows/batch/tag', data)
  },

  // 批量删除
  batchDelete(flowIds: string[]) {
    return request.post('/flows/batch/delete', { flow_ids: flowIds })
  },

  // 导出
  export(flowIds: string[]) {
    return request.post('/flows/batch/export', { flow_ids: flowIds }, {
      responseType: 'blob'
    })
  },

  // 全文搜索
  search(params: FlowSearchParams) {
    return request.post('/flows/search', params)
  }
}
